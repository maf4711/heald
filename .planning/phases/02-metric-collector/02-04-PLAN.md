Create MetricsStore at the app level and pass it to HealdService. The MetricsStore is the shared state that all collectors and downstream consumers use.

```swift
import ArgumentParser
import ServiceLifecycle
import OSLog

struct HealdApp: AsyncParsableCommand {
    static let version = "0.2.0"  // Bump from 0.1.0 — Phase 2 adds metric collection

    static let configuration = CommandConfiguration(
        commandName: "heald",
        abstract: "Self-healing macOS system daemon",
        version: version

        Logger.lifecycle.info("heald \(Self.version) starting")

        let store = MetricsStore()
        let service = HealdService(store: store)
        let serviceGroup = ServiceGroup(
            services: [service],

            logger: .init(label: "com.heald.daemon")
        )

        try await serviceGroup.run()
        Logger.lifecycle.info("heald stopped cleanly")
    }
}
```

**HealdService.swift changes:**

Replace the idle skeleton with active collector wiring. HealdService receives MetricsStore and creates all collector instances. It runs them concurrently using `withTaskGroup` or by listing them as child services.

The ServiceLifecycle pattern: HealdService is itself a Service in the top-level ServiceGroup. Inside its `run()`, it creates a child ServiceGroup containing all collectors + debug writer. This way, graceful shutdown propagates from the top ServiceGroup through HealdService to all child services.

```swift
import ServiceLifecycle
import OSLog

struct HealdService: Service {
    let store: MetricsStore

    func run() async throws {
        Logger.lifecycle.info("HealdService running — starting metric collectors")

        let cpuCollector     = CPUCollector(store: store)
        let ramCollector     = RAMCollector(store: store)
        let diskCollector    = DiskCollector(store: store)
        let processCollector = ProcessCollector(store: store)
        let debugWriter      = DebugStatusWriter(store: store)

        let collectorGroup = ServiceGroup(
            services: [cpuCollector, ramCollector, diskCollector, processCollector, debugWriter],
            logger: .init(label: "com.heald.collectors")
        )

        try await collectorGroup.run()
    }
}
```

This replaces the `Task.sleep(for: .seconds(Int.max))` idle skeleton entirely. The child ServiceGroup runs until graceful shutdown is triggered, at which point each collector's `onGracefulShutdown` callback fires.

**Key considerations:**
- The child `ServiceGroup` does NOT have its own `gracefulShutdownSignals` — only the top-level ServiceGroup in HealdApp catches SIGTERM/SIGINT. The child group receives shutdown propagated from the parent.
- Version bump to 0.2.0 reflects Phase 2 functionality.

**DebugStatusWriter.swift:**

```swift
import Foundation
import ServiceLifecycle
import OSLog

struct DebugStatusWriter: Service {
    let store: MetricsStore

    func run() async throws {
        var shutdown = false
        try await withGracefulShutdownHandler {
            repeat {
                try await writeStatus()
                try await Task.sleep(for: .seconds(5))
            } while !shutdown
        } onGracefulShutdown: {
            shutdown = true
            // Clean up status file on shutdown
            try? FileManager.default.removeItem(atPath: "/tmp/heald-status.json")
            Logger.collector.info("Debug status file removed on shutdown")
        }
    }

    private func writeStatus() async throws {
        let cpu = await store.cpu
        let ram = await store.ram
        let disk = await store.disk
        let processes = await store.processes

        // Build a JSON-serializable dictionary
        let status: [String: Any] = [
            "timestamp": ISO8601DateFormatter().string(from: Date()),
            "cpu": [
                "overall_percent": round(cpu.overall * 1000) / 10,  // e.g. 23.4%
                "core_count": cpu.perCore.count,
                "per_core_percent": cpu.perCore.map { round($0 * 1000) / 10 }
            ],
            "ram": [
                "used_gb": round(ram.used / 1_073_741_824 * 100) / 100,
                "wired_gb": round(ram.wired / 1_073_741_824 * 100) / 100,
                "compressed_gb": round(ram.compressed / 1_073_741_824 * 100) / 100,
                "swap_used_mb": round(ram.swapUsed / 1_048_576 * 100) / 100,
                "swap_total_mb": round(ram.swapTotal / 1_048_576 * 100) / 100,
                "pressure_level": ram.pressureLevel
            ],
            "disk": [
                "volumes": disk.volumes.map { vol in
                    [
                        "name": vol.volumeName,
                        "mount_point": vol.mountPoint,
                        "total_gb": round(Double(vol.totalBytes) / 1_073_741_824 * 100) / 100,
                        "free_gb": round(Double(vol.freeBytes) / 1_073_741_824 * 100) / 100
                    ] as [String: Any]
                },
                "smart": disk.smart.map { s in
                    [
                        "bsd_name": s.bsdName,
                        "status": s.status,
                        "temperature_c": s.temperature as Any,
                        "percentage_used": s.percentageUsed as Any
                    ] as [String: Any]
                }
            ],
            "processes": [
                "top_cpu": processes.byCPU.prefix(5).map { p in
                    ["pid": p.pid, "name": p.name, "cpu_percent": p.cpuPercent, "ram_mb": round(Double(p.ramBytes) / 1_048_576 * 10) / 10, "system": p.isSystem] as [String: Any]
                },
                "top_ram": processes.byRAM.prefix(5).map { p in
                    ["pid": p.pid, "name": p.name, "cpu_percent": p.cpuPercent, "ram_mb": round(Double(p.ramBytes) / 1_048_576 * 10) / 10, "system": p.isSystem] as [String: Any]
                }
            ]
        ]

        let data = try JSONSerialization.data(withJSONObject: status, options: [.prettyPrinted, .sortedKeys])
        try data.write(to: URL(fileURLWithPath: "/tmp/heald-status.json"), options: .atomic)
    }
}
```

**Key details:**
- Writes atomically (`.atomic` option) — readers never see partial JSON
- Overwrites each cycle — no disk growth
- Shows top 5 processes (not all 25) in the debug file to keep it readable
- Human-friendly units: GB for RAM/disk, MB for swap, percentage for CPU
- Cleans up `/tmp/heald-status.json` on graceful shutdown
- Usage: `cat /tmp/heald-status.json | python3 -m json.tool` or `watch cat /tmp/heald-status.json`
